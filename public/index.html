import express from "express"; import fetch from "node-fetch"; import dotenv from "dotenv"; import { randomUUID } from "node:crypto"; dotenv.config(); const app = express(); const PORT = process.env.PORT || 3000; const API_MP = "https://api.mercadopago.com"; const ORIGEM_PERMITIDA = "https://onlinecartomantes-dotcom.github.io"; // ========================= // CORS - COMUNICAÇÃO COM GITHUB PAGES // ========================= app.use((req, res, next) => { res.setHeader( "Access-Control-Allow-Origin", ORIGEM_PERMITIDA ); res.setHeader( "Access-Control-Allow-Methods", "GET, POST, OPTIONS" ); res.setHeader( "Access-Control-Allow-Headers", "Content-Type, Authorization" ); if (req.method === "OPTIONS") { return res.sendStatus(204); } next(); }); app.use(express.json()); // ========================= // VERIFICAR CONFIGURAÇÃO // ========================= if (!process.env.MP_TOKEN) { console.error("ERRO: MP_TOKEN não está configurado no Render."); } // ========================= // ROTA INICIAL - TESTAR API // ========================= app.get("/", (req, res) => { res.json({ online: true, mensagem: "API de pagamentos online", endpoints: [ "POST /create-payment", "GET /check-payment/:id", "POST /webhook" ] }); }); // ========================= // CRIAR PAGAMENTO PIX // ========================= app.post("/create-payment", async (req, res) => { try { if (!process.env.MP_TOKEN) { return res.status(500).json({ error: "MP_TOKEN não configurado no Render." }); } const valor = Number(req.body?.transaction_amount); const descricao = String(req.body?.description || "Pagamento PIX").trim(); if (!Number.isFinite(valor) || valor <= 0) { return res.status(400).json({ error: "O valor do pagamento é inválido." }); } console.log("Solicitação de pagamento recebida:", { transaction_amount: valor, description: descricao }); const corpo = { transaction_amount: valor, payment_method_id: "pix", description: descricao, payer: { email: "marciodoxosse@gmail.com" } }; if (process.env.WEBHOOK_URL) { corpo.notification_url = process.env.WEBHOOK_URL; } const response = await fetch( `${API_MP}/v1/payments`, { method: "POST", headers: { "Content-Type": "application/json", "Authorization": `Bearer ${process.env.MP_TOKEN}`, "X-Idempotency-Key": randomUUID() }, body: JSON.stringify(corpo) } ); const texto = await response.text(); let dados; try { dados = JSON.parse(texto); } catch { dados = { message: "Resposta inválida do Mercado Pago." }; } console.log("Resposta Mercado Pago:", { httpStatus: response.status, paymentId: dados.id || null, status: dados.status || null, message: dados.message || null, cause: dados.cause || null }); if (!response.ok) { return res.status(response.status).json({ error: "Erro no Mercado Pago", details: dados }); } return res.status(200).json(dados); } catch (err) { console.error("Erro interno ao criar pagamento:", err); return res.status(500).json({ error: "Erro interno no servidor." }); } }); // ========================= // CONSULTAR STATUS DO PAGAMENTO // ========================= app.get("/check-payment/:id", async (req, res) => { try { if (!process.env.MP_TOKEN) { return res.status(500).json({ error: "MP_TOKEN não configurado no Render." }); } const paymentId = req.params.id; const response = await fetch( `${API_MP}/v1/payments/${encodeURIComponent(paymentId)}`, { headers: { "Authorization": `Bearer ${process.env.MP_TOKEN}` } } ); const dados = await response.json(); console.log("Consulta de pagamento:", { id: dados.id || paymentId, status: dados.status || null }); return res.status(response.status).json(dados); } catch (err) { console.error("Erro ao consultar pagamento:", err); return res.status(500).json({ error: "Erro ao verificar pagamento." }); } }); // ========================= // WEBHOOK MERCADO PAGO // ========================= app.post("/webhook", async (req, res) => { // Confirma o recebimento da notificação. res.sendStatus(200); try { console.log("Webhook recebido:", req.body); if (!process.env.MP_TOKEN) { console.error("MP_TOKEN não configurado."); return; } const tipo = req.body?.type || req.query?.type; const paymentId = req.body?.data?.id || req.query?.["data.id"]; if (tipo !== "payment" || !paymentId) { console.log("Notificação sem ID de pagamento."); return; } const response = await fetch( `${API_MP}/v1/payments/${encodeURIComponent(paymentId)}`, { headers: { "Authorization": `Bearer ${process.env.MP_TOKEN}` } } ); const paymentData = await response.json(); if (!response.ok) { console.error( "Erro ao consultar pagamento do webhook:", paymentData ); return; } console.log("Status real do pagamento:", { id: paymentData.id, status: paymentData.status, amount: paymentData.transaction_amount }); if (paymentData.status === "approved") { console.log("Pagamento aprovado:", paymentData.id); // Aqui você pode liberar o serviço, // registrar o pagamento no banco de dados // ou enviar uma confirmação ao cliente. } } catch (err) { console.error("Erro ao processar webhook:", err); } }); // ========================= // ROTA NÃO ENCONTRADA // ========================= app.use((req, res) => { res.status(404).json({ error: "Rota não encontrada.", path: req.path }); }); // ========================= // INICIAR SERVIDOR // ========================= app.listen(PORT, () => { console.log(`Servidor rodando na porta ${PORT}`); });
